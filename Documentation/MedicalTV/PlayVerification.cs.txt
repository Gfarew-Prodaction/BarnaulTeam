using System;
using System.IO;
using System.Linq;
using System.Reflection;
using HurricaneVR.Framework.Core.UI;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine;
using UnityEngine.Rendering.Universal;
using UnityEngine.UI;

public static class MedicalTVVerification
{
 static int phase,checks;static double due;static string report="";
 static void Check(bool condition,string message){checks++;if(!condition)throw new Exception(message);}
 static object Get(object o,string name)=>o.GetType().GetField(name,BindingFlags.NonPublic|BindingFlags.Instance).GetValue(o);
 static void Call(object o,string name,params object[] values)=>o.GetType().GetMethod(name,BindingFlags.NonPublic|BindingFlags.Instance).Invoke(o,values);
 static Button Button(Canvas c,string title)=>c.GetComponentsInChildren<Button>().First(b=>b.GetComponentInChildren<Text>().text==title);
 static void Page(MedicalTVScreen tv,Canvas c)
 {
  Check(c&&tv.CurrentPage==c&&c.gameObject.activeInHierarchy,"current page active");Check(c.transform.parent==tv.Screen,"physical screen parent");
  var rect=(RectTransform)c.transform;var size=Vector3.Scale(rect.sizeDelta,c.transform.lossyScale);
  Check(size.x<=tv.ScreenSize.x+.001f&&size.y<=tv.ScreenSize.y+.001f,"canvas fits display");
  Check(Vector3.Distance(c.transform.position,tv.Screen.position)<.001f,"canvas flush");Check(HVRInputModule.Instance.UICanvases.Contains(c),"HVR canvas registered");
 }
 static void Shot(MedicalTVScreen tv,string name,bool room=false)
 {
  Directory.CreateDirectory("output/medical-tv");var go=new GameObject("TV preview");var camera=go.AddComponent<Camera>();camera.enabled=false;camera.nearClipPlane=.02f;camera.farClipPlane=20;
  camera.GetUniversalAdditionalCameraData().renderPostProcessing=true;
  camera.transform.position=room?new Vector3(-.65f,1.62f,1.15f):tv.Screen.position-tv.Screen.forward*2.25f;
  camera.transform.LookAt(room?new Vector3(.7f,1.65f,-1.8f):tv.Screen.position);camera.fieldOfView=room?65:50;
  var avatars=UnityEngine.Object.FindObjectsByType<Renderer>(FindObjectsSortMode.None).Where(r=>r.transform.root.name.Contains("Rig")).ToArray();foreach(var r in avatars)r.forceRenderingOff=true;
  ClinicFurnitureSetup.SaveCamera(camera,"output/medical-tv/"+name+".jpg");foreach(var r in avatars)if(r)r.forceRenderingOff=false;UnityEngine.Object.DestroyImmediate(go);
 }
 public static void Run()
 {
  checks=0;report="";
  foreach(string name in new[]{"SampleScene","PracticeScene"})
  {
   EditorSceneManager.OpenScene("Assets/Scenes/"+name+".unity");var tv=UnityEngine.Object.FindAnyObjectByType<MedicalTVScreen>();Check(tv,"TV saved "+name);Check(tv.ScreenSize.x>1.7f&&tv.ScreenSize.y>.85f,"TV readable size");
   Check(tv.GetComponentsInChildren<MeshCollider>().Length==4,"TV colliders");Check(tv.GetComponentsInChildren<MeshRenderer>().Count(r=>r.enabled)==6,"TV visible mesh count");
   Check(ClinicFurnitureSetup.BoundsOf(tv.gameObject).min.z>=-2.51f,"mount stays inside room");
   foreach(var r in tv.GetComponentsInChildren<MeshRenderer>().Where(r=>r.enabled))foreach(var m in r.sharedMaterials){Check(m&&m.shader.name=="Universal Render Pipeline/Lit","URP material");Check(m.GetTexture("_MetallicGlossMap"),"packed PBR map");}
  }
  EditorSceneManager.OpenScene("Assets/Scenes/SampleScene.unity");phase=0;EditorApplication.update+=Tick;EditorApplication.isPlaying=true;due=EditorApplication.timeSinceStartup+4;
 }
 static void Tick()
 {
  if(EditorApplication.isCompiling||EditorApplication.timeSinceStartup<due)return;
  try
  {
   if(phase==0&&EditorApplication.isPlaying)
   {
    var tv=UnityEngine.Object.FindAnyObjectByType<MedicalTVScreen>();var nav=UnityEngine.Object.FindAnyObjectByType<StudyModePanel>();var study=(Canvas)Get(nav,"_panel");Page(tv,study);Shot(tv,"study");Shot(tv,"study-room",true);
    Check(HVRInputModule.Instance.MaxDistance>=5,"pointer reaches wall TV");
    Button(study,"Управление").onClick.Invoke();var help=tv.CurrentPage;Page(tv,help);Check(!study.gameObject.activeSelf,"help replaces navigation");Shot(tv,"controls-1");
    Button(help,"Далее →").onClick.Invoke();Check(help.GetComponentsInChildren<Text>().Any(t=>t.text.Contains("2 из 3")),"help page forward");Shot(tv,"controls-2");Button(help,"Далее →").onClick.Invoke();Shot(tv,"controls-3");
    Button(help,"Закрыть").onClick.Invoke();Page(tv,study);
    Button(study,"Обучение").onClick.Invoke();phase=1;due=EditorApplication.timeSinceStartup+1;
   }
   else if(phase==1&&EditorApplication.isPlaying)
   {
    var tv=UnityEngine.Object.FindAnyObjectByType<MedicalTVScreen>();var guide=UnityEngine.Object.FindAnyObjectByType<VRLearningGuide>();var canvas=(Canvas)Get(guide,"_canvas");Page(tv,canvas);Check(guide.StepCount==25,"25 steps retained");Shot(tv,"learning");
    for(int i=0;i<guide.StepCount;i++){Call(guide,"Enter",i);Page(tv,canvas);Check(canvas.GetComponentsInChildren<Button>().Length==5,"guide buttons");}
    Shot(tv,"learning-last");Button(canvas,"Управление").onClick.Invoke();var help=tv.CurrentPage;Check(!canvas.gameObject.activeSelf,"help replaces guide");Button(help,"Закрыть").onClick.Invoke();Page(tv,canvas);
    Call(guide,"Enter",0);var pos=canvas.transform.position;var head=Camera.main.transform;var headpos=head.position;head.position+=Vector3.right*.5f;Call(guide,"Reposition");Check(Vector3.Distance(pos,canvas.transform.position)<.001f,"TV page stays fixed while player moves");head.position=headpos;
    Call(guide,"Enter",24);Call(guide,"Next");phase=2;due=EditorApplication.timeSinceStartup+.5;
   }
   else if(phase==2&&EditorApplication.isPlaying)
   {
    var tv=UnityEngine.Object.FindAnyObjectByType<MedicalTVScreen>();Check(tv.CurrentPage.name=="Study · Navigation","completion restores study");Check(!UnityEngine.Object.FindAnyObjectByType<VRLearningGuide>(),"guide removed");report+="Study, paged controls, all 25 learning steps and return navigation passed.\n";EditorApplication.isPlaying=false;phase=3;due=EditorApplication.timeSinceStartup+1;
   }
   else if(phase==3&&!EditorApplication.isPlaying)
   {
    EditorSceneManager.OpenScene("Assets/Scenes/PracticeScene.unity");EditorApplication.isPlaying=true;phase=4;due=EditorApplication.timeSinceStartup+4;
   }
   else if(phase==4&&EditorApplication.isPlaying)
   {
    var tv=UnityEngine.Object.FindAnyObjectByType<MedicalTVScreen>();var session=UnityEngine.Object.FindAnyObjectByType<PracticeSession>();var canvas=(Canvas)Get(session,"_panel");Page(tv,canvas);Shot(tv,"practice");Shot(tv,"practice-room",true);
    Button(canvas,"Управление").onClick.Invoke();var help=tv.CurrentPage;Page(tv,help);Button(help,"Закрыть").onClick.Invoke();Page(tv,canvas);
    session.Begin(PracticeDifficulty.Advanced,PracticeSet.All);Check(session.ReferencePiece&&session.PlacedCount==1,"reference still works");Button(canvas,"Заново").onClick.Invoke();Check(session.Mistakes==0&&session.PlacedCount==1,"restart button");
    Check(canvas.GetComponentsInChildren<Text>().Any(t=>t.text.Contains("Установлено")),"progress shown on TV");report+="Practice progress, help, restart and anchored reference passed.\n";EditorApplication.isPlaying=false;phase=5;due=EditorApplication.timeSinceStartup+1;
   }
   else if(phase==5&&!EditorApplication.isPlaying)
   {
    EditorSceneManager.OpenScene("Assets/Scenes/SampleScene.unity");Directory.CreateDirectory("Documentation/MedicalTV");File.WriteAllText("Documentation/MedicalTV/validation.txt","PASS "+checks+" assertions\n"+report+"Physical Quest headset not tested.\n");File.WriteAllText("/tmp/tv-tests.txt","PASS "+checks+"\n"+report);EditorApplication.update-=Tick;
   }
  }
  catch(Exception e){File.WriteAllText("/tmp/tv-tests.txt","FAIL "+checks+"\n"+e);EditorApplication.update-=Tick;EditorApplication.isPlaying=false;Debug.LogException(e);}
 }
}
